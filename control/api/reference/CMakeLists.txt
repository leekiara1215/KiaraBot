find_package(Boost 1.86 CONFIG REQUIRED)
add_library(kiarabot_control STATIC ${CMAKE_CURRENT_LIST_DIR}/../src/http_mapper.cpp ${CMAKE_CURRENT_LIST_DIR}/../src/http_server.cpp ${CMAKE_CURRENT_LIST_DIR}/../src/configuration.cpp ${CMAKE_CURRENT_LIST_DIR}/../src/event_stream.cpp)
target_include_directories(kiarabot_control PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../include)
target_link_libraries(kiarabot_control PUBLIC kiarabot_bridge PRIVATE Boost::headers)
target_compile_options(kiarabot_control PRIVATE
  $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall -Wextra -Wpedantic>)
add_executable(kiarabot-control-credentials ${CMAKE_CURRENT_LIST_DIR}/../src/credentials.cpp)
target_link_libraries(kiarabot-control-credentials PRIVATE kiarabot_bridge)
target_link_libraries(kiarabot-reference-core PRIVATE kiarabot_control)
add_executable(kiarabot-event-stream-test ${CMAKE_CURRENT_LIST_DIR}/../tests/event_stream_test.cpp)
target_link_libraries(kiarabot-event-stream-test PRIVATE kiarabot_control)
add_test(NAME control_event_stream COMMAND kiarabot-event-stream-test)
if(KIARABOT_ENABLE_INFRASTRUCTURE AND EXISTS "${CMAKE_CURRENT_LIST_DIR}/../tests/.venv/bin/python")
  add_test(NAME control_http_websocket_acceptance
    COMMAND "${CMAKE_CURRENT_LIST_DIR}/../tests/.venv/bin/python" -m pytest -q "${CMAKE_CURRENT_SOURCE_DIR}/tests")
  set_tests_properties(control_http_websocket_acceptance PROPERTIES
    TIMEOUT 90 ENVIRONMENT "KIARABOT_ACCEPTANCE_BUILD=${CMAKE_BINARY_DIR}/reference")
endif()
add_executable(kiarabot-control-mock ${CMAKE_CURRENT_LIST_DIR}/../tests/mock_server.cpp)
target_link_libraries(kiarabot-control-mock PRIVATE kiarabot_control)

get_property(REFERENCE_TESTS DIRECTORY PROPERTY TESTS)
foreach(TEST_NAME IN LISTS REFERENCE_TESTS)
  set_tests_properties(${TEST_NAME} PROPERTIES LABELS "reference")
endforeach()
get_property(REFERENCE_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(TARGET_NAME IN LISTS REFERENCE_TARGETS)
  set_property(TARGET ${TARGET_NAME} PROPERTY KIARABOT_REFERENCE TRUE)
endforeach()
