cmake_minimum_required(VERSION 3.24)
project(kiarabot_inner_contracts LANGUAGES CXX)
enable_testing()
add_executable(inner_contracts_test foundation_test.cpp)
target_compile_features(inner_contracts_test PRIVATE cxx_std_20)
target_include_directories(inner_contracts_test PRIVATE ../../include)
if(MSVC)
  target_compile_options(inner_contracts_test PRIVATE /W4)
else()
  target_compile_options(inner_contracts_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME inner_contracts COMMAND inner_contracts_test)
set_tests_properties(inner_contracts PROPERTIES LABELS "architecture;p01b")

add_executable(inner_invariants_test invariants_test.cpp)
target_compile_features(inner_invariants_test PRIVATE cxx_std_20)
target_include_directories(inner_invariants_test PRIVATE ../../include)
if(MSVC)
  target_compile_options(inner_invariants_test PRIVATE /W4)
else()
  target_compile_options(inner_invariants_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME inner_invariants COMMAND inner_invariants_test)
set_tests_properties(inner_invariants PROPERTIES LABELS "architecture;p01b")
