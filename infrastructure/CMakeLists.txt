cmake_minimum_required(VERSION 3.24)
project(kiarabot_infrastructure LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Protobuf CONFIG REQUIRED)
find_program(KIARABOT_PROTOC protoc REQUIRED)
set(PROTO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../proto)
set(PROTO_OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(PROTO_NAMES common events actions plugins)
set(PROTO_INPUTS)
set(PROTO_SOURCES)
set(PROTO_HEADERS)
foreach(NAME IN LISTS PROTO_NAMES)
  list(APPEND PROTO_INPUTS ${PROTO_ROOT}/kiarabot/v1/${NAME}.proto)
  list(APPEND PROTO_SOURCES ${PROTO_OUTPUT}/kiarabot/v1/${NAME}.pb.cc)
  list(APPEND PROTO_HEADERS ${PROTO_OUTPUT}/kiarabot/v1/${NAME}.pb.h)
endforeach()
add_custom_command(OUTPUT ${PROTO_SOURCES} ${PROTO_HEADERS}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${PROTO_OUTPUT}
  COMMAND ${KIARABOT_PROTOC} --proto_path=${PROTO_ROOT} --cpp_out=${PROTO_OUTPUT} ${PROTO_INPUTS}
  DEPENDS ${PROTO_INPUTS} VERBATIM)
add_library(infrastructure_proto STATIC ${PROTO_SOURCES})
target_include_directories(infrastructure_proto SYSTEM PUBLIC ${PROTO_OUTPUT})
target_link_libraries(infrastructure_proto PUBLIC protobuf::libprotobuf)
add_library(kiarabot_infrastructure STATIC src/file_storage.cpp)
target_include_directories(kiarabot_infrastructure PUBLIC include ../core/include)
target_link_libraries(kiarabot_infrastructure PUBLIC infrastructure_proto)
target_compile_options(kiarabot_infrastructure PRIVATE
  $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall -Wextra -Wpedantic>
  $<$<CXX_COMPILER_ID:MSVC>:/W4>)
enable_testing()
add_executable(file_storage_test tests/file_storage_test.cpp)
target_link_libraries(file_storage_test PRIVATE kiarabot_infrastructure)
add_test(NAME file_storage COMMAND file_storage_test)
